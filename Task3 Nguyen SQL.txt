#3) Která kategorie potravin zdražuje nejpomaleji (je u ní nejnižší percentuální meziroční nárůst)?

WITH growth AS (
    SELECT 
        t1.food_code,
        t1.food_name,
        ROUND(((t2.value / t1.value) - 1) * 100, 2) AS yoy_growth
    FROM t_huu_viet_nguyen_project_SQL_primary_final t1
    JOIN t_huu_viet_nguyen_project_SQL_primary_final t2
        ON t2.payroll_year = t1.payroll_year + 1
        AND t1.food_code = t2.food_code
)

SELECT 
    food_code,
    food_name,
    ROUND(AVG(yoy_growth), 2) AS avg_yoy_growth_percentage
FROM growth
GROUP BY food_code, food_name
ORDER BY avg_yoy_growth_percentage;
